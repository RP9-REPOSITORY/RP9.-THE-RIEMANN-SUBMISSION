P.B.F. FULL POSITIONAL GENEALOGY — 100B VALIDATOR
==================================================

Author: Anton Wallin
Framework: RP9 / Relational Primacy Framework
Version: 4.1.0-100B-POSITIONAL-GENEALOGY

Function
--------
Executes and validates the K.7 P.B.F. locked-observation genealogy mechanically
through 100,000,000,000.

The generator preserves the operational basis:
    Fixed Observation Field
    -> Sequential Addition
    -> Genealogical Coincidence
    -> New Closure / Reproduced Position

The generator does NOT use:
    - primality tests
    - modulo classification
    - divisibility queries for candidate classification
    - factor search
    - p^2 start rule
    - an externally supplied later-prime list

Memory architecture
-------------------
The full domain is NOT allocated at once. It is processed as consecutive
segments. Active genealogical references carry their next sequential endpoint
from one segment into the next. Thus domain size is decoupled from RAM size.

Validation architecture
-----------------------
For every segment:
1. Generate the locked P.B.F. observation field.
2. Apply internally generated genealogical references by sequential endpoint
   continuation.
3. Produce the P.B.F. closure bitmap for that segment.
4. Only then build an independent conventional reference bitmap.
5. Compare both bitmaps byte-for-byte.

A scale can PASS only if every segment through that scale is an exact positional
match. Equal counts alone are NOT sufficient.

Submission status
-----------------
PASS requires exact byte-for-byte positional identity, zero false positives,
zero false negatives, and exact observation-count agreement throughout the
executed domain.

Automatic checkpoints preserve the completed sequential state during the long
execution. If the process is restarted, the same file resumes from the latest
valid completed segment.
"""

def now_iso() -> str:

def write_json(path: Path, payload) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload, indent=2, ensure_ascii=False), encoding="utf-8")


def write_json_atomic(path: Path, payload) -> None:
    """Write a recoverable state file without exposing a partial checkpoint."""
    temporary = path.with_suffix(path.suffix + ".tmp")
    write_json(temporary, payload)
    os.replace(temporary, path)

def fixed_observation_count_exact(limit: int) -> int:
    """Exact cardinality of the locked field generated by 2, 3, then 6k±1."""
    if limit < 2:
        return 0
    count = 1
    if limit >= 3:
        count += 1
    if limit >= 5:
        count += (limit + 1) // 6
        count += max(0, (limit - 1) // 6)
    return count

def build_locked_reference_field(limit: int) -> bytearray:
    field = bytearray(limit + 1)
    if limit >= 2:
        field[2] = 1
    if limit >= 3:
        field[3] = 1
    center = 6
    while center <= limit + 1:
        left = center - 1
        right = center + 1
        if 5 <= left <= limit:
            field[left] = 1
        if 5 <= right <= limit:
            field[right] = 1
        center += 6
    return field


def project_internal_genealogy(reference: int, limit: int, field: bytearray, hit: bytearray) -> None:
    x = reference + reference
    while x <= limit:
        if field[x]:
            hit[x] = 1
        x += reference


def generate_internal_references(limit: int) -> List[int]:
    """
    Generate all P.B.F. closures through `limit` with the original K7 mechanism.

    This is NOT an external prime list. It is generated internally from the same
    fixed field + sequential addition + coincidence rules used by the original
    benchmark.
    """
    field = build_locked_reference_field(limit)
    hit = bytearray(limit + 1)
    closures: List[int] = []

    for p in (2, 3):
        if p <= limit and field[p]:
            closures.append(p)
            project_internal_genealogy(p, limit, field, hit)

    position = 5
    while position <= limit:
        if field[position]:
            if not hit[position]:
                closures.append(position)
                project_internal_genealogy(position, limit, field, hit)
        position += 1

    return closures

def fill_locked_observation_segment(
    low: int,
    high: int,
    next_left: int,
    next_right: int,
) -> Tuple[bytearray, int, int, int]:
    """
    Build exactly the same locked observation field inside [low, high].

    next_left / next_right carry the next sequential 6-step endpoints across
    segment boundaries. No modulo classification is used.
    """
    size = high - low + 1
    field = bytearray(size)

    if low <= 2 <= high:
        field[2 - low] = 1
    if low <= 3 <= high:
        field[3 - low] = 1

    # center-1 stream: 5, 11, 17, ...
    if next_left < low:
        while next_left < low:
            next_left += 6
    if next_left <= high:
        start = next_left - low
        count = len(range(start, size, 6))
        field[start:size:6] = b"\x01" * count
        next_left += count * 6

    # center+1 stream: 7, 13, 19, ...
    if next_right < low:
        while next_right < low:
            next_right += 6
    if next_right <= high:
        start = next_right - low
        count = len(range(start, size, 6))
        field[start:size:6] = b"\x01" * count
        next_right += count * 6

    return field, int(sum(field)), next_left, next_right


def apply_genealogy_segment(
    field: bytearray,
    low: int,
    high: int,
    references: List[int],
    next_hits: List[int],
) -> None:
    """
    Continue every internally generated genealogy through this segment.

    Each genealogy starts at 2p and thereafter advances only by +p. Its next
    endpoint is persisted between segments. Slice clearing is only a storage
    optimization for the same sequential endpoint series.
    """
    size = len(field)

    for i, p in enumerate(references):
        x = next_hits[i]
        if x > high:
            continue

        # The prior segment leaves x at the first endpoint beyond its boundary,
        # so x is normally already >= low. The loop is a safety-preserving guard.
        while x < low:
            x += p

        if x > high:
            next_hits[i] = x
            continue

        local_start = x - low
        count = len(range(local_start, size, p))
        field[local_start:size:p] = b"\x00" * count
        next_hits[i] = x + count * p


# ---------------------------------------------------------------------------
# EXTERNAL POST-GENERATION REFERENCE (INDEPENDENT)
# ---------------------------------------------------------------------------

def simple_reference_primes(limit: int) -> List[int]:
    if limit < 2:
        return []
    sieve = bytearray(b"\x01") * (limit + 1)
    sieve[0:2] = b"\x00\x00"
    root = isqrt(limit)
    for p in range(2, root + 1):
        if sieve[p]:
            start = p * p
            sieve[start:limit + 1:p] = b"\x00" * len(range(start, limit + 1, p))
    return [i for i in range(2, limit + 1) if sieve[i]]


def build_external_reference_segment(low: int, high: int, base_primes: List[int]) -> bytearray:
    """Independent conventional reference, created only after P.B.F. generation."""
    size = high - low + 1
    ref = bytearray(b"\x01") * size

    if low == 0:
        if size > 0:
            ref[0] = 0
        if size > 1:
            ref[1] = 0
    elif low == 1:
        ref[0] = 0

    root = isqrt(high)
    for p in base_primes:
        if p > root:
            break
        start = p * p
        if start < low:
            # External reference only: conventional segment alignment.
            start = ((low + p - 1) // p) * p
        if start <= high:
            local = start - low
            ref[local:size:p] = b"\x00" * len(range(local, size, p))

    return ref

def save_checkpoint(
    path: Path,
    limits: List[int],
    segment_size: int,
    reference_count: int,
    state: Dict[str, object],
) -> None:
    write_json_atomic(path, {
        "program": OFFICIAL_CODE_TITLE,
        "version": VERSION,
        "status": "RUNNING",
        "saved_at": now_iso(),
        "limits": limits,
        "segment_size": segment_size,
        "reference_count": reference_count,
        "state": state,
    })


def load_checkpoint(
    path: Path,
    limits: List[int],
    segment_size: int,
    reference_count: int,
) -> Optional[Dict[str, object]]:
    if not path.exists():
        return None
    payload = json.loads(path.read_text(encoding="utf-8"))
    expected = {
        "program": OFFICIAL_CODE_TITLE,
        "version": VERSION,
        "limits": limits,
        "segment_size": segment_size,
        "reference_count": reference_count,
    }
    for key, value in expected.items():
        if payload.get(key) != value:
            raise ValueError(f"Checkpoint mismatch for {key}: {payload.get(key)!r} != {value!r}")
    if payload.get("status") != "RUNNING" or not isinstance(payload.get("state"), dict):
        raise ValueError("Checkpoint does not contain a resumable RUNNING state.")
    return payload["state"]


def execute_full_genealogy(
    scale_limits: List[int],
    segment_size: int,
    checkpoint_path: Optional[Path] = None,
) -> Dict[str, object]:
    limits = sorted(set(int(x) for x in scale_limits if int(x) >= 2))
    if not limits:
        raise ValueError("At least one limit >= 2 is required.")
    if segment_size < 1_000_000:
        raise ValueError("segment_size must be >= 1,000,000")

    max_limit = max(limits)
    reference_horizon = isqrt(max_limit)

    print("=" * 96)
    print("P.B.F. FULL POSITIONAL GENEALOGY — VALIDATION")
    print("=" * 96)
    print(f"Maximum domain          : {max_limit:,}")
    print(f"Segment size            : {segment_size:,}")
    print(f"Internal ref horizon    : {reference_horizon:,}")
    print("Generating internal P.B.F. references...")

    t_refs = time.perf_counter()
    references = generate_internal_references(reference_horizon)
    refs_duration = time.perf_counter() - t_refs
    next_hits = [p + p for p in references]

    print(f"Internal references     : {len(references):,}")
    print(f"Reference generation    : {refs_duration:.4f} s")
    print("Generating independent post-generation base reference...")

    t_ext = time.perf_counter()
    external_base = simple_reference_primes(reference_horizon)
    ext_duration = time.perf_counter() - t_ext
    print(f"External base refs      : {len(external_base):,}")
    print(f"External base time      : {ext_duration:.4f} s")
    print("-" * 96)

    resumed = False
    prior_elapsed = 0.0
    saved_state = (
        load_checkpoint(
            checkpoint_path, limits, segment_size, len(references)
        )
        if checkpoint_path is not None else None
    )

    if saved_state is None:
        target_index = 0
        low = 0
        next_left = 5
        next_right = 7
        cumulative_observations = 0
        cumulative_closures = 0
        cumulative_false_positive = 0
        cumulative_false_negative = 0
        cumulative_exact = True
        first_deviation_position = None
        results: List[Dict[str, object]] = []
        trace: List[Dict[str, object]] = []
        segment_number = 0
    else:
        resumed = True
        target_index = int(saved_state["target_index"])
        low = int(saved_state["next_low"])
        next_left = int(saved_state["next_left"])
        next_right = int(saved_state["next_right"])
        cumulative_observations = int(saved_state["cumulative_observations"])
        cumulative_closures = int(saved_state["cumulative_closures"])
        cumulative_false_positive = int(saved_state["cumulative_false_positive"])
        cumulative_false_negative = int(saved_state["cumulative_false_negative"])
        cumulative_exact = bool(saved_state["cumulative_exact"])
        first_deviation_position = saved_state["first_deviation_position"]
        results = list(saved_state["results"])
        trace = list(saved_state["trace"])
        segment_number = int(saved_state["segment_number"])
        prior_elapsed = float(saved_state["elapsed_seconds"])
        restored_hits = saved_state["next_hits"]
        if len(restored_hits) != len(next_hits):
            raise ValueError("Checkpoint next-hit state has the wrong reference count.")
        next_hits = [int(x) for x in restored_hits]
        print(f"Resuming checkpoint      : position {low:,}")
        print(f"Completed segments       : {segment_number:,}")

    next_target = limits[target_index]
    global_start = time.perf_counter()

    while low <= max_limit:
        natural_high = min(max_limit, low + segment_size - 1)
        high = min(natural_high, next_target)
        segment_number += 1
        seg_start_time = time.perf_counter()

        # 1. P.B.F. generation first.
        pbf_field, obs_count, next_left, next_right = fill_locked_observation_segment(
            low, high, next_left, next_right
        )
        apply_genealogy_segment(pbf_field, low, high, references, next_hits)
        pbf_count = int(sum(pbf_field))
        cumulative_observations += obs_count
        cumulative_closures += pbf_count

        # 2. Independent reference only after P.B.F. segment is complete.
        reference_field = build_external_reference_segment(low, high, external_base)
        reference_count = int(sum(reference_field))

        segment_exact = pbf_field == reference_field
        fp = 0
        fn = 0
        first_local_deviation = None

        # Exact positional mismatch diagnostics are evaluated only if needed.
        if not segment_exact:
            for offset, (a, b) in enumerate(zip(pbf_field, reference_field)):
                if a and not b:
                    fp += 1
                    if first_local_deviation is None:
                        first_local_deviation = low + offset
                elif b and not a:
                    fn += 1
                    if first_local_deviation is None:
                        first_local_deviation = low + offset

        cumulative_false_positive += fp
        cumulative_false_negative += fn
        cumulative_exact = cumulative_exact and segment_exact
        if first_deviation_position is None and first_local_deviation is not None:
            first_deviation_position = first_local_deviation

        seg_duration = time.perf_counter() - seg_start_time

        trace.append({
            "segment": segment_number,
            "low": low,
            "high": high,
            "observation_positions": obs_count,
            "pbf_closures": pbf_count,
            "external_reference": reference_count,
            "exact_positional_match": segment_exact,
            "false_positives": fp,
            "false_negatives": fn,
            "duration_seconds": round(seg_duration, 6),
        })

        del reference_field
        del pbf_field

        if high == next_target:
            elapsed = prior_elapsed + time.perf_counter() - global_start
            expected_observations = fixed_observation_count_exact(next_target)
            observation_formula_match = cumulative_observations == expected_observations
            status = (
                "PASS"
                if cumulative_exact
                and cumulative_false_positive == 0
                and cumulative_false_negative == 0
                and observation_formula_match
                else "FAIL"
            )

            row = {
                "limit": next_target,
                "observation_positions": cumulative_observations,
                "expected_observation_positions": expected_observations,
                "observation_formula_match": observation_formula_match,
                "pbf_new_closures": cumulative_closures,
                "reproduced_positions": cumulative_observations - cumulative_closures,
                "false_positives": cumulative_false_positive,
                "false_negatives": cumulative_false_negative,
                "exact_positional_match": cumulative_exact,
                "first_deviation_position": first_deviation_position,
                "elapsed_seconds": round(elapsed, 4),
                "status": status,
            }
            results.append(row)

            print(
                f"LIMIT {next_target:>15,} | OBS {cumulative_observations:>15,} | "
                f"CLOSURES {cumulative_closures:>15,} | EXACT {str(cumulative_exact):<5} | "
                f"STATUS {status}"
            )
            target_index += 1
            if target_index >= len(limits):
                break
            next_target = limits[target_index]

        low = high + 1

        if (
            checkpoint_path is not None
            and segment_number % CHECKPOINT_INTERVAL_SEGMENTS == 0
        ):
            checkpoint_elapsed = prior_elapsed + time.perf_counter() - global_start
            save_checkpoint(
                checkpoint_path,
                limits,
                segment_size,
                len(references),
                {
                    "target_index": target_index,
                    "next_low": low,
                    "next_left": next_left,
                    "next_right": next_right,
                    "cumulative_observations": cumulative_observations,
                    "cumulative_closures": cumulative_closures,
                    "cumulative_false_positive": cumulative_false_positive,
                    "cumulative_false_negative": cumulative_false_negative,
                    "cumulative_exact": cumulative_exact,
                    "first_deviation_position": first_deviation_position,
                    "results": results,
                    "trace": trace,
                    "segment_number": segment_number,
                    "elapsed_seconds": round(checkpoint_elapsed, 6),
                    "next_hits": next_hits,
                },
            )
            print(
                f"CHECKPOINT {segment_number:>6,} | NEXT POSITION {low:>15,} | "
                f"ELAPSED {checkpoint_elapsed:>10.1f} s"
            )

    total_duration = prior_elapsed + time.perf_counter() - global_start
    all_pass = all(r["status"] == "PASS" for r in results)
    return {
        "metadata": {
            "repository": "RP9. RH. Z0",
            "repository_branch": "RP9. RIEMANN SUBMISSION",
            "python_code_id": PYTHON_CODE_ID,
            "filesystem_name": FILE_NAME,
            "official_code_title": OFFICIAL_CODE_TITLE,
            "repository_position": "PRIMARY SUBMISSION — SUBJECT OF EXAMINATION AND REVIEW",
            "version": VERSION,
            "generated_at": now_iso(),
            "python_version": sys.version,
            "segment_size": segment_size,
            "maximum_limit": max_limit,
            "internal_reference_horizon": reference_horizon,
            "internal_reference_count": len(references),
            "internal_reference_generation_seconds": round(refs_duration, 6),
            "external_base_reference_count": len(external_base),
            "external_base_generation_seconds": round(ext_duration, 6),
            "total_execution_seconds": round(total_duration, 4),
            "resumed_from_checkpoint": resumed,
            "checkpoint_interval_segments": CHECKPOINT_INTERVAL_SEGMENTS,
        },
        "execution_architecture": {
            "pbf_generator": "internally generated genealogical references",
            "external_base_reference": "independently generated before segment execution",
            "external_comparison_bitmap": "constructed after the corresponding P.B.F. segment",
            "data_flow": "no external reference data is passed into the P.B.F. generator",
            "comparison": "exact byte-for-byte positional comparison",
        },
        "results": results,
        "summary": {
            "global_status": "PASS" if all_pass else "FAIL",
            "first_deviation_position": first_deviation_position,
            "maximum_executed_genealogy_status": "PASS" if all_pass else "FAIL",
            "maximum_executed_genealogy_limit": max_limit,
        },
        "trace": trace,
    }

def build_csv(payload: Dict[str, object]) -> List[str]:

        "limit,observation_positions,pbf_new_closures,reproduced_positions,false_positives,false_negatives,exact_positional_match,elapsed_seconds,status"
    ]
    for r in payload["results"]:
        lines.append(
            ",".join([
                str(r["limit"]), str(r["observation_positions"]), str(r["pbf_new_closures"]),
                str(r["reproduced_positions"]), str(r["false_positives"]), str(r["false_negatives"]),
                str(r["exact_positional_match"]), str(r["elapsed_seconds"]), str(r["status"]),
            ])
        )
    return lines


def build_control_log(payload: Dict[str, object]) -> List[str]:
    s = payload["summary"]

    def shown(value: object, grouped: bool = False) -> str:
        if value is None:
            return "NOT_EXECUTED"
        if grouped and isinstance(value, int):
            return f"{value:,}"
        return str(value)

    lines = [
        "P.B.F. FULL POSITIONAL GENEALOGY — CONTROL LOG",
        "=" * 88,
        f"Python Code ID: {PYTHON_CODE_ID}",
        f"Official Code Title: {OFFICIAL_CODE_TITLE}",
        f"File: {FILE_NAME}",
        f"Version: {VERSION}",
        f"Timestamp: {payload['metadata']['generated_at']}",
        f"Global Status: {s['global_status']}",
        f"Maximum Executed Genealogy Status: {s['maximum_executed_genealogy_status']}",
        f"First Deviation Position: {s['first_deviation_position'] if s['first_deviation_position'] is not None else 'NONE'}",
        "",
        "SCALE BREAKDOWN",
        "-" * 88,
    ]
    for r in payload["results"]:
        lines += [
            f"LIMIT: {r['limit']:,}",
            f"  Observation positions: {r['observation_positions']:,}",
            f"  P.B.F. closures: {shown(r['pbf_new_closures'], grouped=True)}",
            f"  Reproduced positions: {shown(r['reproduced_positions'], grouped=True)}",
            f"  False positives: {shown(r['false_positives'])}",
            f"  False negatives: {shown(r['false_negatives'])}",
            f"  Exact positional match: {shown(r['exact_positional_match'])}",
            f"  Status: {r['status']}",
            "",
        ]
    return lines


def write_outputs(output_dir: Path, payload: Dict[str, object]) -> Dict[str, Path]:
    output_dir.mkdir(parents=True, exist_ok=True)
    paths = {
        "data_json": output_dir / DATA_FILE,
        "final_report": output_dir / FINAL_REPORT_FILE,
        "data_sheet": output_dir / DATA_SHEET_FILE,
        "csv": output_dir / CSV_FILE,
        "control_log": output_dir / CONTROL_FILE,
        "execution_log": output_dir / EXECUTION_FILE,
    }
    write_json(paths["data_json"], payload)
    write_text(paths["final_report"], build_report(payload))
    write_text(paths["data_sheet"], build_data_sheet(payload))
    write_text(paths["csv"], build_csv(payload))
    write_text(paths["control_log"], build_control_log(payload))
    write_text(paths["execution_log"], [
        "P.B.F. FULL POSITIONAL GENEALOGY — EXECUTION LOG",
        "=" * 88,
        f"Timestamp: {now_text()}",
        f"Python Code ID: {PYTHON_CODE_ID}",
        f"Official Code Title: {OFFICIAL_CODE_TITLE}",
        f"File: {FILE_NAME}",
        f"Version: {VERSION}",
        f"Maximum limit: {payload['metadata']['maximum_limit']:,}",
        f"Global status: {payload['summary']['global_status']}",
        f"Maximum executed genealogy status: {payload['summary']['maximum_executed_genealogy_status']}",
        f"Total execution: {payload['metadata']['total_execution_seconds']} s",
    ])
    return paths

# CLI
# ---------------------------------------------------------------------------

def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="Validate the P.B.F. full positional genealogy through the stated domain."
    )
    parser.add_argument(
        "--limits", type=int, nargs="+", default=DEFAULT_SCALES,
        help="Mechanical milestones through the final 100B domain.",
    )
    parser.add_argument(
        "--segment-size", type=int, default=DEFAULT_SEGMENT_SIZE,
        help=f"Integer positions per memory segment. Default: {DEFAULT_SEGMENT_SIZE:,}.",
    )
    parser.add_argument("--output-dir", type=str, default=None)
    return parser.parse_args()


def main() -> int:
    args = parse_args()
    limits = sorted(set(args.limits))
    if max(limits) > MAXIMUM_SUPPORTED_LIMIT:
        raise ValueError(
            "Mechanical limits must not exceed the supported 100B domain."
        )

    root = base_dir()
    output_dir = (
        Path(args.output_dir).expanduser().resolve()
        if args.output_dir else root / OUTPUT_DIR_NAME
    )
    checkpoint_path = output_dir / CHECKPOINT_FILE

    try:
        # Create visible output immediately and record that execution started.
        output_dir.mkdir(parents=True, exist_ok=True)
        (output_dir / ERROR_FILE).unlink(missing_ok=True)
        write_text(output_dir / EXECUTION_FILE, [
            "P.B.F. FULL POSITIONAL GENEALOGY — LIVE EXECUTION LOG",
            "=" * 88,
            f"Started: {now_text()}",
            f"Python Code ID: {PYTHON_CODE_ID}",
            f"Official Code Title: {OFFICIAL_CODE_TITLE}",
            f"Version: {VERSION}",
            f"Execution milestone limits: {', '.join(f'{x:,}' for x in limits)}",
            "Status: RUNNING",
        ])
        payload = execute_full_genealogy(
            limits, args.segment_size, checkpoint_path=checkpoint_path
        )
        paths = write_outputs(output_dir, payload)
        checkpoint_path.unlink(missing_ok=True)
        print("=" * 96)
        print(f"GLOBAL STATUS              : {payload['summary']['global_status']}")
        print(f"MAXIMUM EXECUTED DOMAIN    : {payload['summary']['maximum_executed_genealogy_limit']:,}")
        print(f"POSITIONAL GENEALOGY       : {payload['summary']['maximum_executed_genealogy_status']}")
        first_deviation = payload["summary"]["first_deviation_position"]
        print(f"FIRST DEVIATION POSITION   : {first_deviation if first_deviation is not None else 'NONE'}")
        print(f"OUTPUT DIRECTORY           : {output_dir}")
        print("=" * 96)
        return 0 if payload["summary"]["global_status"] == "PASS" else 1
    except Exception as exc:
        output_dir.mkdir(parents=True, exist_ok=True)
        error_path = output_dir / ERROR_FILE
        write_text(error_path, [
            "P.B.F. FULL POSITIONAL GENEALOGY — ERROR LOG",
            "=" * 88,
            f"Timestamp: {now_text()}",
            f"Python Code ID: {PYTHON_CODE_ID}",
            f"Official Code Title: {OFFICIAL_CODE_TITLE}",
            f"Version: {VERSION}",
            f"Error type: {type(exc).__name__}",
            f"Error: {exc}",
            "",
            traceback.format_exc(),
        ])
        print(f"CRITICAL ERROR: {exc}")
        print(f"Error log: {error_path}")
        return 1


if __name__ == "__main__":
    exit_code = main()
    if os.name == "nt" and sys.stdin.isatty():
        try:
            input("Press Enter to close...")
        except EOFError:
            pass
    raise SystemExit(exit_code)
